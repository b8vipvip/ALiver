// Add these lines to the official ConsoleApplication_source callback after
// downloading the current pure_PipeSDK.zip from Douyin Open Platform.
// The exact CreatePipeClient signature remains owned by the official SDK/demo;
// this patch only handles EVENT_MESSAGE data after the SDK has connected.

#include "aliver_event_sink.h"

AliverEventSink g_aliver_sink;

// During plugin startup, before subscribing to OPEN_LIVE_DATA:
std::string sink_error;
g_aliver_sink.Start(
    L"ALiverDouyinCollector.exe",
    L"douyin_collector.json",
    &sink_error);

// Keep the official subscription call:
// method: x.subscribeEvent
// params: {"eventName":"OPEN_LIVE_DATA","timestamp":<milliseconds>}

// Inside IPipeClient::SetCallback(...):
// The official SDK callback receives `data` and `size` when type == EVENT_MESSAGE.
if (type == PipeSDK::EVENT_MESSAGE && data != nullptr && size > 0) {
    const std::string json(data, size);
    if (json.find("\"eventName\":\"OPEN_LIVE_DATA\"") != std::string::npos ||
        json.find("\"eventName\": \"OPEN_LIVE_DATA\"") != std::string::npos) {
        std::string forward_error;
        if (!g_aliver_sink.SendJsonLine(data, size, &forward_error)) {
            // Write forward_error to the plugin log. Do not terminate the PipeSDK callback thread.
        }
    }
}

// When receiving EVENT_DISCONNECTED / OPEN_WIN_CLOSE:
g_aliver_sink.Stop();
